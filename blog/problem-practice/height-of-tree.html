maxLength = 0

def findHightOfTree(head, length):
    if head is None:
        print(length)
        return length
    maxHeight = 0
    if hasattr(head, "left"):
        lhight = findHightOfTree(head.left, length)
        maxHeight = lhight + 1
    if hasattr(head, "right"):
        rhight = findHightOfTree(head.right, length)
        if maxHeight - 1 < rhight:
            maxHeight = rhight+1

    print(maxHeight)
    return maxHeight;
